================================================================
1. СТРАТЕГИЯ БЭКАПА
================================================================

Делаем "домашний каталог как dotfiles-проект":
  /root  →  ~/mima_01_bk_workdir/  →  push в mima_01_bk

Принципы:
  • Берём ТОЛЬКО то, что восстанавливается руками за час, и что
    жалко потерять: конфиги, скрипты, документы, настройки.
  • НЕ берём кэши, скачанные модели, виртуальные окружения,
    состояние пакетных менеджеров — это восстанавливается одной
    командой и раздувает репо.
  • Ключи и пароли НЕ бэкапим (см. раздел 6).
  • Один коммит = один снимок с датой в сообщении, чтобы видеть
    историю изменений.

3. ПОДГОТОВКА РАБОЧЕГО КАТАЛОГА (одноразово)
================================================================

  # 3.1. Выбираем, где держать "зеркало" домашней папки.
  # Лучше НЕ в /root, чтобы случайно не запушить лишнее.
  # Я кладу в /opt — он есть на сервере.
  sudo mkdir -p /opt/mima_01_bk_workdir
  sudo chown root:root /opt/mima_01_bk_workdir
  cd /opt/mima_01_bk_workdir

  # 3.2. Клонируем пустой/существующий репозиторий.
  #     Через https — сработает rewrite на SSH (см. раздел 0).
  git clone https://github.com/kuznetsovrost-svg/mima_01_bk.git
  cd mima_01_bk

  # 3.3. Создаём .gitignore под наш whitelist+blacklist.
  #     Содержимое — в Приложении A ниже.

  # 3.4. Говорим git-у "наш корень" — это КОПИЯ /root,
  #     а не сам /root. Создаём зеркало.
  #     ВАРИАНТ A (рекомендую): rsync в work-каталог с фильтром,
  #     а git add/commit уже оттуда. Так никогда не запушишь
  #     лишнее: то, чего нет в work-каталоге, в git не попадёт.
  #
  #     Команды — в разделе 4.

4. ОДНОРАЗОВАЯ СИНХРОНИЗАЦИЯ (первый бэкап)
================================================================

  # 4.1. Создаём .gitignore (Приложение A).
  # 4.2. rsync из /root в work-каталог.
  #     Исключения — те же, что blacklist в .gitignore, чтобы
  #     случайно не заливать гигабайты.

  cd /opt/mima_01_bk_workdir/mima_01_bk

  rsync -a --delete \
        --exclude='.git' \
        --exclude='.cache' \
        --exclude='.npm' \
        --exclude='.hermes' \
        --exclude='.ssh' \
        --exclude='.gnupg' \
        --exclude='.local/lib' \
        --exclude='.local/share/virtualenvs' \
        --exclude='.local/share/Trash' \
        --exclude='.local/share/uv' \
        --exclude='.local/share/uv-python' \
        --exclude='.config/*/Cache' \
        --exclude='.config/*/Caches' \
        --exclude='__pycache__' \
        --exclude='*.pyc' \
        --exclude='.bash_history' \
        --exclude='*.swp' --exclude='*.swo' --exclude='*~' \
        /root/  ./root/

  # Создастся подкаталог ./root/ со всем whitelist-содержимым.
  # Проверь глазами:
  ls -la root/
  du -sh root/    # ожидаю: <100 MB, реально десятки мегабайт

  # 4.3. Не обязательно, но удобно: манифест окружения.
  #     Это поможет восстановить сервер, даже если сам бэкап
  #     окажется старым.
  {
    echo "=== Снимок сервера $(date -Iseconds) ==="
    echo
    echo "## /etc/os-release"
    cat /etc/os-release
    echo
    echo "## uname -a"
    uname -a
    echo
    echo "## dpkg --get-selections (только реально установленное)"
    dpkg --get-selections | awk '$2=="install"{print $1}' \
         | sort > root/manifest/packages.txt
    echo "  /opt/mima_01_bk_workdir/mima_01_bk/root/manifest/packages.txt"
    echo
    echo "## pipx list (если есть)"
    command -v pipx >/dev/null && pipx list || echo "  pipx не установлен"
    echo
    echo "## crontab -l"
    crontab -l > root/manifest/crontab.txt 2>/dev/null \
       && echo "  crontab сохранён в root/manifest/crontab.txt" \
       || echo "  crontab пуст или не настроен"
    echo
    echo "## systemctl list-unit-files --state=enabled"
    systemctl list-unit-files --state=enabled > root/manifest/services.txt 2>/dev/null
    echo "  /opt/mima_01_bk_workdir/mima_01_bk/root/manifest/services.txt"
    echo
    echo "## df -h"
    df -h > root/manifest/disk.txt
  } | tee root/manifest/SNAPSHOT.txt

  # 4.4. git add + commit + push.
  git add -A
  git status               # ОБЯЗАТЕЛЬНО посмотри глазами
  git commit -m "backup: server snapshot $(date -Iseconds)"
  git push -u origin HEAD  # в первый раз с -u, дальше без

  (они безопасны, если это правда *.pub).

  Перед КАЖДЫМ git push делай:
        git status
        git diff --cached --stat
  и убедись, что в списке нет ничего из blacklist.

ПРИЛОЖЕНИЕ B — /usr/local/bin/backup-to-mima01.sh
================================================================

  #!/usr/bin/env bash
  set -euo pipefail

  WORK=/opt/mima_01_bk_workdir/mima_01_bk
  cd "$WORK" || { echo "Нет $WORK"; exit 1; }

  rsync -a --delete \
        --exclude='.git' --exclude='.cache' --exclude='.npm' \
        --exclude='.hermes' --exclude='.ssh' --exclude='.gnupg' \
        --exclude='.local/lib' --exclude='.local/share/virtualenvs' \
        --exclude='.local/share/Trash' --exclude='.local/share/uv' \
        --exclude='.local/share/uv-python' \
        --exclude='.config/*/Cache' --exclude='.config/*/Caches' \
        --exclude='__pycache__' --exclude='*.pyc' \
        --exclude='.bash_history' \
        --exclude='*.swp' --exclude='*.swo' --exclude='*~' \
        /root/  ./root/

  {
    echo "=== Снимок сервера $(date -Iseconds) ==="
    cat /etc/os-release
    uname -a
  } > root/manifest/SNAPSHOT.txt

  if git diff --cached --quiet && \
     git diff --quiet && \
     [ -z "$(git ls-files --others --exclude-standard)" ]; then
    echo "Нечего коммитить: $WORK актуален."
    exit 0
  fi

  git add -A
  git commit -m "backup: server snapshot $(date -Iseconds)"
  git push

  echo "OK: бэкап отправлен."

      Должно быть kuznetsovrost-svg.
