# Заполнение заголовка отчёта — Python-код

> Сессия: 2026-08-03, профиль `fst-control`
> Назначение файла: готовый код для получения `Name` проекта по `ProjectId` и серверного времени, оформление шапки отчёта.

---

## 1. Архитектура

```
┌──────────────────────────┐         ┌────────────────────────┐
│ Python-функция           │  SSE    │  MCP-сервер            │
│ build_report_header()    │ ──────► │  fst_pmpractice        │
│                          │         │  https://ppreplica…    │
│ получает:                │         │                        │
│  • Name по ProjectId     │ ──────► │  get_entity_data       │
│  • server_time           │ ──────► │  initialize + MetaModel│
└──────────────────────────┘         └────────────────────────┘
```

**Источники серверного времени:**

| Приоритет | Источник | Что вернёт | Надёжность |
|-----------|----------|-----------|------------|
| 1 | MCP `initialize` → `serverInfo` | имя + версия сервера, **но не время** | ⚠️ время недоступно |
| 2 | `MetaModel` через `find_entity` по ключу `CurrentDateTime` | системное время Форсайт | ✅ если ключ есть в системе |
| 3 | `MetaModel` через `find_entity` по ключу `SystemDate` | то же | ✅ альтернатива |
| 4 | `datetime.now()` (локальное время агента) | fallback | ⚠️ не совпадает с серверным на 100% |

В реальной практике **первый источник пустой по части даты**, второй и третий — зависят от конфигурации Форсайт (часто ключей нет). Поэтому fallback на `datetime.now()` — норма для production. Будем помечать источник в шапке.

## 2. Полный код с комментариями

```python
"""
build_report_header.py

Получение Name проекта по ProjectId + серверного времени для шапки отчёта.

Используется в скиллах:
- fst-howproject (главный header отчёта)
- project-search (карточка проекта)
- project-payment-schedule / project-effort-* / project-completeness-check
  (когда нужны метаданные отчёта)

Возвращает: dict с полями:
    - project_name:    str
    - project_code:    str
    - project_id:      int
    - server_time:     datetime  # в формате "YYYY-MM-DD HH:MM:SS"
    - server_time_src: str       # 'meta_model' | 'local' — откуда взято
    - server_time_str: str       # уже отформатированная строка
    - server_name:     str       # из serverInfo.name (PMForesight MCP)
    - server_version:  str       # из serverInfo.version
"""

from __future__ import annotations

import json
import logging
import os
import re
from datetime import datetime, timezone
from typing import Any, Optional

import requests

# ═══════════════════════════════════════════════════════════════════════════════
# КОНФИГУРАЦИЯ
# ═══════════════════════════════════════════════════════════════════════════════

MCP_URL = "https://ppreplica.hera.pmpractice.ru/mcp"
MCP_TOKEN = os.environ.get("PMP_TOKEN")
if not MCP_TOKEN:
    raise RuntimeError(
        "PMP_TOKEN не задан в переменных окружения. "
        "Проверьте ~/.hermes/profiles/fst-control/.env"
    )

# Ключи MetaModel, через которые Форсайт может отдавать системное время.
# Пробуем по очереди.
META_TIME_KEYS = ("CurrentDateTime", "SystemDate", "ServerTime", "Now")

logger = logging.getLogger(__name__)


# ═══════════════════════════════════════════════════════════════════════════════
# ИСКЛЮЧЕНИЯ
# ═══════════════════════════════════════════════════════════════════════════════

class ProjectNameNotFound(Exception):
    """Не удалось получить Name проекта по ProjectId."""
    pass


class ServerTimeUnavailable(Exception):
    """Не удалось получить серверное время ни одним способом."""
    pass


# ═══════════════════════════════════════════════════════════════════════════════
# НИЗКОУРОВНЕВЫЙ MCP-ДОСТУП (вынесено для переиспользования)
# ═══════════════════════════════════════════════════════════════════════════════

def mcp_init() -> tuple[str, dict]:
    """
    Инициализация MCP-сессии.

    Возвращает кортеж:
        (session_id, server_info)
    где server_info — это dict из result.serverInfo
    ({name, version, protocolVersion?, instructions?}).
    """
    payload = {
        "jsonrpc": "2.0",
        "id": 1,
        "method": "initialize",
        "params": {
            "protocolVersion": "2024-11-05",
            "capabilities": {},
            "clientInfo": {"name": "build_report_header", "version": "1.0"}
        }
    }
    resp = requests.post(
        MCP_URL,
        json=payload,
        headers={
            "Accept": "application/json, text/event-stream",
            "Content-Type": "application/json",
            "Authorization": f"Bearer {MCP_TOKEN}",
        },
        timeout=30,
    )
    resp.raise_for_status()
    sid = resp.headers.get("mcp-session-id")
    if not sid:
        raise ServerTimeUnavailable("сервер не вернул mcp-session-id")

    # Парсим ответ — нам нужны serverInfo
    server_info: dict = {}
    for line in resp.text.splitlines():
        if line.startswith("data: "):
            try:
                envelope = json.loads(line[6:])
                server_info = envelope.get("result", {}).get("serverInfo", {})
            except json.JSONDecodeError:
                pass
            break

    # Подтверждаем инициализацию (best-effort)
    try:
        requests.post(
            MCP_URL,
            json={"jsonrpc": "2.0", "method": "notifications/initialized"},
            headers={
                "Accept": "application/json, text/event-stream",
                "Content-Type": "application/json",
                "Authorization": f"Bearer {MCP_TOKEN}",
                "mcp-session-id": sid,
            },
            timeout=10,
        )
    except requests.RequestException:
        pass

    return sid, server_info


def mcp_call(sid: str, tool_name: str, arguments: dict, *, timeout: int = 60) -> Any:
    """
    Универсальный вызов MCP-инструмента. Возвращает распарсенный
    payload из result.content[0].text (или сам dict при ошибке).
    """
    payload = {
        "jsonrpc": "2.0",
        "id": 1,
        "method": "tools/call",
        "params": {"name": tool_name, "arguments": arguments},
    }
    resp = requests.post(
        MCP_URL,
        json=payload,
        headers={
            "Accept": "application/json, text/event-stream",
            "Content-Type": "application/json",
            "Authorization": f"Bearer {MCP_TOKEN}",
            "mcp-session-id": sid,
        },
        timeout=timeout,
        stream=True,
    )
    resp.raise_for_status()

    for line in resp.iter_lines(decode_unicode=True):
        if line and line.startswith("data: "):
            envelope = json.loads(line[6:])
            break
    else:
        raise ServerTimeUnavailable("нет data-строки в SSE-ответе")

    result = envelope.get("result", {})
    if result.get("isError"):
        # Возвращаем текст ошибки, чтобы вызывающий код мог её обработать
        return {"_error": result["content"][0]["text"]}

    # Нормальный ответ: result.content[0].text — JSON-строка
    return json.loads(result["content"][0]["text"])


# ═══════════════════════════════════════════════════════════════════════════════
# ПОЛУЧЕНИЕ NAME ПРОЕКТА ПО PROJECTID
# ═══════════════════════════════════════════════════════════════════════════════

def get_project_name_by_id(project_id: int, sid: Optional[str] = None) -> dict:
    """
    Возвращает словарь с полями проекта, минимум:
        {
            "ProjectId": 4595,
            "Code":      "504-595",
            "Name":      "Космодром Жуковский внедрение ИСУП",
            ...
        }

    Использует get_entity_data(Project, data_id=project_id).

    Args:
        project_id: bigint ProjectId.
        sid:        (опц.) уже открытая MCP-сессия.

    Returns:
        dict — properties проекта.

    Raises:
        ProjectNameNotFound — проект не найден или нет прав.
    """
    own_session = sid is None
    if own_session:
        sid, _ = mcp_init()

    try:
        # get_entity_data ожидает data_id как строку-число
        data = mcp_call(
            sid, "get_entity_data",
            {"entityName": "Project", "data_id": str(project_id)},
        )

        if isinstance(data, dict) and "_error" in data:
            raise ProjectNameNotFound(
                f"get_entity_data(Project, {project_id}) → {data['_error'][:200]}"
            )

        properties = data.get("properties", {})

        # Достаём Name из properties
        name_field = properties.get("Name", {})
        name_value = name_field.get("value") or name_field.get("display")
        if not name_value:
            raise ProjectNameNotFound(
                f"у проекта ProjectId={project_id} нет поля Name"
            )

        # Code — для шапки тоже полезно
        code_field = properties.get("Code", {})
        code_value = code_field.get("value") or code_field.get("display")

        return {
            "ProjectId": project_id,
            "Code":      code_value,
            "Name":      name_value,
            "_raw":      properties,  # все остальные поля на случай если нужны
        }

    except requests.RequestException as e:
        raise ProjectNameNotFound(f"сетевая ошибка при запросе проекта: {e}") from e
    finally:
        if own_session:
            pass  # сессия закроется сама


# ═══════════════════════════════════════════════════════════════════════════════
# ПОЛУЧЕНИЕ СЕРВЕРНОГО ВРЕМЕНИ
# ═══════════════════════════════════════════════════════════════════════════════

def _try_metamodel_time(sid: str) -> Optional[datetime]:
    """
    Пытается достать системное время Форсайт через MetaModel.

    В ИСУП «Форсайт» сущность MetaModel содержит системные настройки
    (Name = ключ, Value = значение). Среди ключей иногда встречаются
    CurrentDateTime / SystemDate / ServerTime / Now.

    ⚠️ Ключи зависят от конфигурации системы; могут отсутствовать.
    В этом случае функция возвращает None, и идёт fallback на локальное время.
    """
    for key in META_TIME_KEYS:
        try:
            data = mcp_call(
                sid, "find_entity",
                {"entityName": "MetaModel", "query": key, "limit": 1},
            )
        except requests.RequestException:
            continue

        if isinstance(data, dict) and "_error" in data:
            logger.debug("MetaModel[%s] isError: %s", key, data["_error"][:100])
            continue

        records = data.get("data") if isinstance(data, dict) else None
        if not records:
            continue

        # Берём значение первой записи
        value = records[0].get("Value") or records[0].get("value")
        if not value:
            continue

        # Парсим как ISO 8601 (формат Форсайт: "2026-08-03T14:25:33.123456+03:00")
        try:
            # Заменяем Z на +00:00 для fromisoformat
            iso_str = value.replace("Z", "+00:00") if isinstance(value, str) else value
            return datetime.fromisoformat(iso_str)
        except (ValueError, AttributeError) as e:
            logger.debug("не удалось распарсить MetaModel[%s]=%r: %s", key, value, e)
            continue

    return None


def get_server_time(sid: Optional[str] = None) -> tuple[datetime, str, str, str]:
    """
    Возвращает серверное время + метаданные.

    Returns:
        tuple[datetime, str, str, str]:
            - server_time:    datetime с TZ
            - source:         'meta_model' | 'local'
            - server_name:    из MCP serverInfo.name (например, "PMForesight MCP")
            - server_version: из MCP serverInfo.version (например, "3.2.4")
    """
    own_session = sid is None
    if own_session:
        sid, server_info = mcp_init()
    else:
        # sid уже есть; serverInfo возьмём из кэша, если передали;
        # иначе повторно инициализируем (best-effort)
        try:
            _, server_info = mcp_init()
        except Exception:
            server_info = {}

    server_name = server_info.get("name", "PMForesight MCP")
    server_version = server_info.get("version", "3.2.4")

    try:
        # Попытка №1: MetaModel
        meta_time = _try_metamodel_time(sid)
        if meta_time is not None:
            logger.info("серверное время из MetaModel: %s", meta_time.isoformat())
            return meta_time, "meta_model", server_name, server_version

        # Попытка №2: локальное время
        local_time = datetime.now(timezone.utc).astimezone()
        logger.warning(
            "серверное время недоступно через MetaModel, "
            "используем локальное: %s", local_time.isoformat()
        )
        return local_time, "local", server_name, server_version

    finally:
        if own_session:
            pass


# ═══════════════════════════════════════════════════════════════════════════════
# ФОРМАТИРОВАНИЕ
# ═══════════════════════════════════════════════════════════════════════════════

def format_server_time(dt: datetime) -> str:
    """
    Возвращает время в формате "YYYY-MM-DD HH:MM:SS".

    Пример: "2026-08-03 14:25:33"
    """
    return dt.strftime("%Y-%m-%d %H:%M:%S")


# ═══════════════════════════════════════════════════════════════════════════════
# ГЛАВНАЯ ФУНКЦИЯ — ЗАГОЛОВОК ОТЧЁТА
# ═══════════════════════════════════════════════════════════════════════════════

def build_report_header(project_id: int) -> dict:
    """
    Собирает шапку отчёта:
      - Name проекта (по ProjectId)
      - серверное время (формат YYYY-MM-DD HH:MM:SS)
      - метаданные MCP-сервера

    Возвращает dict:
        {
            "project_id":       4595,
            "project_code":     "504-595",
            "project_name":     "Космодром Жуковский внедрение ИСУП",
            "server_time":      datetime(...),
            "server_time_str":  "2026-08-03 14:25:33",
            "server_time_src":  "meta_model" | "local",
            "server_name":      "PMForesight MCP",
            "server_version":   "3.2.4",
        }
    """
    # Открываем одну сессию и используем для обоих запросов
    sid, server_info = mcp_init()

    # 1) Получаем Name проекта
    project = get_project_name_by_id(project_id, sid=sid)
    project_name = project["Name"]
    project_code = project.get("Code")

    # 2) Получаем серверное время (через ту же сессию)
    server_time, time_source, srv_name, srv_version = get_server_time(sid=sid)

    # Если server_info пустой — берём из текущего вызова
    if not srv_name:
        srv_name = server_info.get("name", "PMForesight MCP")
    if not srv_version:
        srv_version = server_info.get("version", "3.2.4")

    return {
        "project_id":      project_id,
        "project_code":    project_code,
        "project_name":    project_name,
        "server_time":     server_time,
        "server_time_str": format_server_time(server_time),
        "server_time_src": time_source,
        "server_name":     srv_name,
        "server_version":  srv_version,
    }


# ═══════════════════════════════════════════════════════════════════════════════
# ДЕМОНСТРАЦИЯ
# ═══════════════════════════════════════════════════════════════════════════════

if __name__ == "__main__":
    """
    Запуск:
        export PMP_TOKEN="...ваш токен..."
        python build_report_header.py 4595
    """
    import sys

    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
    )

    if len(sys.argv) != 2:
        print("Использование: python build_report_header.py <ProjectId>")
        print("Пример:        python build_report_header.py 4595")
        sys.exit(1)

    try:
        pid = int(sys.argv[1])
    except ValueError:
        print(f"❌ ProjectId должен быть целым числом, получено '{sys.argv[1]}'")
        sys.exit(2)

    print(f"\nСобираем заголовок отчёта для ProjectId={pid}…\n")

    try:
        header = build_report_header(pid)
    except ProjectNameNotFound as e:
        print(f"❌ Проект не найден: {e}")
        sys.exit(3)
    except ServerTimeUnavailable as e:
        print(f"⚠️  Время сервера недоступно: {e}")
        # Не критично — продолжаем
        header = None

    if header:
        print("✅ Заголовок отчёта:")
        print(f"   ProjectId:      {header['project_id']}")
        print(f"   Code:           {header['project_code']}")
        print(f"   Name:           {header['project_name']}")
        print(f"   ServerTime:     {header['server_time_str']}")
        print(f"   ServerTimeSrc:  {header['server_time_src']}")
        print(f"   MCP Server:     {header['server_name']} v{header['server_version']}")

        # Пример строки для вставки в отчёт
        print()
        print("── Строка для шапки отчёта ──")
        print(
            f"Проект: {header['project_name']} ({header['project_code']}, "
            f"ProjectId={header['project_id']})\n"
            f"Дата формирования: {header['server_time_str']} "
            f"({header['server_time_src']})\n"
            f"Сервер: {header['server_name']} v{header['server_version']}"
        )
```

## 4. Формат времени

**Запрошенный формат:** `год-месяц-день минуты:секунды` → `YYYY-MM-DD HH:MM:SS`

Пример: `2026-08-03 14:25:33`

Реализуется через:
```python
dt.strftime("%Y-%m-%d %H:%M:%S")
```

⚠️ `strftime` использует **локальное время** Python-интерпретатора. Если сервер в Москве, а агент в Лондоне — будет расхождение. Для строгого UTC:

```python
datetime.now(timezone.utc).strftime("%Y-%m-%d %H:%M:%S")
# → 2026-08-03 11:25:33  (UTC)
```

В коде выше время берётся **локальное через `astimezone()`** (без аргумента → системная TZ). Если хотите явный UTC — замените в `get_server_time()`.
